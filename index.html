/** * Apps Script Web App untuk menerima laporan dari form "Lapor Kendala Depo" * dan menulis baris baru ke Google Sheet ini, serta menyimpan file bukti * dukung ke folder Google Drive. * * CARA PASANG: * 1. Buka spreadsheet Anda -> menu "Extensions" / "Ekstensi" -> "Apps Script" * 2. Hapus semua kode default di editor yang terbuka * 3. Tempel seluruh isi file ini * 4. Klik ikon Simpan (atau Ctrl+S) * 5. Klik "Deploy" -> "New deployment" * - Klik ikon gerigi di samping "Select type" -> pilih "Web app" * - Description: bebas, misal "Form Kendala Depo" * - Execute as: "Me" * - Who has access: "Anyone" * 6. Klik "Deploy", lalu izinkan (Authorize) akses yang diminta Google * (akan muncul peringatan "Google hasn't verified this app" -> klik * "Advanced" -> "Go to (nama project) (unsafe)" -> ini aman karena * ini skrip milik Anda sendiri) * 7. Salin "Web app URL" yang muncul (diakhiri dengan /exec) * 8. Kirim URL tersebut ke saya (Claude) agar saya sambungkan ke form * * PENTING kalau Anda menempelkan ulang kode versi baru ini (menggantikan * versi lama): setelah Save, klik "Deploy" -> "Manage deployments" -> * klik ikon pensil pada deployment yang ada -> ubah "Version" jadi * "New version" -> klik "Deploy". Kalau hanya menyimpan kode tanpa * membuat versi baru, Web App masih memakai kode lama. * * PENTING KHUSUS UPDATE INI (struktur sheet berubah, ada kolom ID/RTL baru): * Setelah menempel kode ini dan deploy ulang, buka menu "Run" di editor * Apps Script, pilih fungsi `migrateSheetToV2` dari dropdown di sebelah * tombol Run, lalu klik Run SATU KALI. Ini akan memindahkan data lama * Anda (No, Nama Depo, Uraian, Bukti Dukung, Waktu) ke struktur kolom * yang baru tanpa menghapus data. Aman dijalankan berkali-kali (kalau * sheet sudah berformat baru, fungsi ini akan berhenti otomatis tanpa * mengubah apa-apa). * * KALAU MUNCUL ERROR "Akses ditolak: DriveApp" saat upload file: * 1. Pastikan TARGET_FOLDER_ID di bawah ini benar (ambil dari URL folder * Drive tujuan: https://drive.google.com/drive/folders/INI_ID_NYA) * 2. Pastikan akun Google yang dipakai untuk Deploy (Execute as: Me) * adalah pemilik folder tsb, atau minimal punya akses Editor. * 3. Setelah mengubah kode, buat "New version" saat deploy, lalu jalankan * fungsi apa saja sekali dari editor supaya izin akses Drive diminta * ulang (re-authorize). */ // Ganti dengan ID folder Google Drive tujuan penyimpanan file bukti dukung. // ID ini diambil dari bagian akhir URL folder Drive Anda: // https://drive.google.com/drive/folders/INI_ID_NYA var TARGET_FOLDER_ID = '113sZz3IC99czp54f3x-wHZdEZxAFeR1n'; // Urutan kolom di sheet (baris 1 = header) var HEADER = ['ID', 'No', 'Nama Depo', 'Tanggal Kendala', 'Uraian Kendala', 'Bukti Dukung', 'RTL', 'Tanggal Selesai RTL', 'File RTL', 'Waktu Lapor']; var COL = { ID: 1, NO: 2, NAMA_DEPO: 3, TANGGAL: 4, URAIAN: 5, BUKTI: 6, RTL: 7, SELESAI_RTL: 8, FILE_RTL: 9, WAKTU: 10 }; function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action || 'create'; if (action === 'delete') return handleDelete_(data); if (action === 'update') return handleUpdate_(data); return handleCreate_(data); } catch (err) { return jsonOut_({ status: 'error', message: err.message }); } } function doGet(e) { try { var sheet = getSheet_(); var lastRow = sheet.getLastRow(); var result = []; if (lastRow > 1) { var values = sheet.getRange(2, 1, lastRow - 1, HEADER.length).getValues(); for (var i = 0; i < values.length; i++) { var row = values[i]; var bukti = row[COL.BUKTI - 1] ? String(row[COL.BUKTI - 1]).split('\n').filter(function (x) { return x; }) : []; var fileRtlRaw = row[COL.FILE_RTL - 1] ? String(row[COL.FILE_RTL - 1]).split('\n').filter(function (x) { return x; }) : []; var waktuVal = row[COL.WAKTU - 1]; var waktuStr = waktuVal instanceof Date ? Utilities.formatDate(waktuVal, Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm') : String(waktuVal || ''); result.push({ id: row[COL.ID - 1], no: row[COL.NO - 1], namaDepo: row[COL.NAMA_DEPO - 1], tanggal: row[COL.TANGGAL - 1], uraian: row[COL.URAIAN - 1], files: bukti.map(parseFileEntry_), rtl: row[COL.RTL - 1], selesaiRtl: row[COL.SELESAI_RTL - 1], fileRtl: fileRtlRaw.map(parseFileEntry_), waktu: waktuStr }); } } return jsonOut_({ status: 'ok', data: result }); } catch (err) { return jsonOut_({ status: 'error', message: err.message, data: [] }); } } /* ---------------- CREATE ---------------- */ function handleCreate_(data) { var lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (lockErr) { return jsonOut_({ status: 'error', message: 'Server sedang sibuk, coba lagi sebentar lagi.' }); } var rowIndex; try { var sheet = getSheet_(); var nextNo = sheet.getLastRow(); // baris terisi saat ini (termasuk header) = No urut baris data berikutnya var waktu = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm'); var id = Utilities.getUuid(); // PENTING: simpan data teks DULU, sebelum menyentuh file, supaya // laporan tetap tersimpan walau upload file di bawah gagal. sheet.appendRow([id, nextNo, data.namaDepo, formatDateStr_(data.tanggal), data.uraian, '', '', '', '', waktu]); rowIndex = sheet.getLastRow(); } catch (err) { return jsonOut_({ status: 'error', message: err.message }); } finally { lock.releaseLock(); } // Baru proses file bukti dukung secara terpisah, di luar lock, supaya // laporan yang sudah tersimpan di atas tidak ikut hilang kalau upload gagal. var uploadResult = uploadFilesToFolder_(data.files || []); if (uploadResult.entries.length > 0) { writeCellSafely_(rowIndex, COL.BUKTI, uploadResult.entries.join('\n')); } var responseObj = { status: 'ok', id: id }; if (uploadResult.errors.length > 0) responseObj.fileWarnings = uploadResult.errors; return jsonOut_(responseObj); } /* ---------------- DELETE ---------------- */ function handleDelete_(data) { if (!data.id) return jsonOut_({ status: 'error', message: 'ID laporan tidak ditemukan.' }); var lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (lockErr) { return jsonOut_({ status: 'error', message: 'Server sedang sibuk, coba lagi sebentar lagi.' }); } try { var sheet = getSheet_(); var rowIndex = findRowById_(sheet, data.id); if (rowIndex === -1) { return jsonOut_({ status: 'error', message: 'Laporan tidak ditemukan (mungkin sudah dihapus sebelumnya).' }); } sheet.deleteRow(rowIndex); return jsonOut_({ status: 'ok' }); } catch (err) { return jsonOut_({ status: 'error', message: err.message }); } finally { lock.releaseLock(); } } /* ---------------- UPDATE (RTL) ---------------- */ function handleUpdate_(data) { if (!data.id) return jsonOut_({ status: 'error', message: 'ID laporan tidak ditemukan.' }); var lock = LockService.getScriptLock(); try { lock.waitLock(20000); } catch (lockErr) { return jsonOut_({ status: 'error', message: 'Server sedang sibuk, coba lagi sebentar lagi.' }); } var rowIndex; try { var sheet = getSheet_(); rowIndex = findRowById_(sheet, data.id); if (rowIndex === -1) { return jsonOut_({ status: 'error', message: 'Laporan tidak ditemukan (mungkin sudah dihapus).' }); } if (typeof data.rtl === 'string') { sheet.getRange(rowIndex, COL.RTL).setValue(data.rtl); } if (typeof data.selesaiRtl === 'string') { sheet.getRange(rowIndex, COL.SELESAI_RTL).setValue(formatDateStr_(data.selesaiRtl)); } } catch (err) { return jsonOut_({ status: 'error', message: err.message }); } finally { lock.releaseLock(); } var responseObj = { status: 'ok' }; var filesArray = data.files || []; if (filesArray.length > 0) { var uploadResult = uploadFilesToFolder_(filesArray); if (uploadResult.entries.length > 0) { var existing = readCellSafely_(rowIndex, COL.FILE_RTL); var combined = existing ? (existing + '\n' + uploadResult.entries.join('\n')) : uploadResult.entries.join('\n'); writeCellSafely_(rowIndex, COL.FILE_RTL, combined); } if (uploadResult.errors.length > 0) responseObj.fileWarnings = uploadResult.errors; } return jsonOut_(responseObj); } /* ---------------- HELPERS ---------------- */ function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; if (sheet.getLastRow() === 0) { sheet.appendRow(HEADER); sheet.getRange(1, 1, 1, HEADER.length).setFontWeight('bold'); } return sheet; } function findRowById_(sheet, id) { var lastRow = sheet.getLastRow(); if (lastRow < 2) return -1; var ids = sheet.getRange(2, COL.ID, lastRow - 1, 1).getValues(); for (var i = 0; i < ids.length; i++) { if (ids[i][0] === id) return i + 2; } return -1; } function writeCellSafely_(rowIndex, col, value) { var lock = LockService.getScriptLock(); try { lock.waitLock(20000); var sheet = getSheet_(); sheet.getRange(rowIndex, col).setValue(value); } catch (e) { // kalau gagal tulis (jarang terjadi), abaikan supaya request utama tetap berhasil } finally { try { lock.releaseLock(); } catch (e2) { /* abaikan */ } } } function readCellSafely_(rowIndex, col) { try { var sheet = getSheet_(); return sheet.getRange(rowIndex, col).getValue(); } catch (e) { return ''; } } /** * Fungsi BANTU DIAGNOSA. Jalankan sekali dari editor Apps Script (pilih * "testDriveAccess" di dropdown sebelah tombol Run, lalu klik Run), lalu * buka menu "Execution log" / "Log Eksekusi" (atau tekan Ctrl+Enter) untuk * membaca hasilnya. Ini akan menunjukkan persis kenapa upload file gagal: * apakah folder tidak ditemukan, beda pemilik, atau akses tidak cukup. */ function testDriveAccess() { Logger.log('Akun yang menjalankan script ini: ' + Session.getActiveUser().getEmail()); try { var folder = DriveApp.getFolderById(TARGET_FOLDER_ID); Logger.log('OK - Folder ditemukan: "' + folder.getName() + '"'); try { Logger.log('Pemilik folder: ' + folder.getOwner().getEmail()); } catch (ownerErr) { Logger.log('(Tidak bisa membaca pemilik folder: ' + ownerErr.message + ')'); } try { var testFile = folder.createFile(Utilities.newBlob('test', 'text/plain', 'test-akses-hapus-saja.txt')); Logger.log('BERHASIL membuat file test di folder ini: ' + testFile.getUrl()); testFile.setTrashed(true); Logger.log('File test sudah dihapus lagi. Akses folder ini SUDAH BENAR, seharusnya upload bukti dukung bisa jalan.'); } catch (createErr) { Logger.log('GAGAL membuat file di folder ini: ' + createErr.message); Logger.log('Ini berarti akun di atas HANYA punya akses "Lihat" (Viewer) ke folder, bukan "Editor". ' + 'Perbaikan: buka folder ini di Drive, klik "Share/Bagikan", ubah peran akun tsb jadi Editor - ' + 'atau lebih simpel, buat folder BARU langsung dari akun di atas lalu ganti TARGET_FOLDER_ID dengan ID folder baru itu.'); } } catch (folderErr) { Logger.log('GAGAL membuka folder dengan ID "' + TARGET_FOLDER_ID + '": ' + folderErr.message); Logger.log('Kemungkinan: ID folder salah/typo, folder sudah dihapus/dipindah ke trash, ' + 'atau akun di atas sama sekali tidak punya akses ke folder ini. ' + 'Perbaikan tercepat: buat folder baru dari akun ini, salin ID barunya dari URL, ganti TARGET_FOLDER_ID.'); } } function uploadFilesToFolder_(filesArray) { var entries = []; var errors = []; if (!filesArray || filesArray.length === 0) return { entries: entries, errors: errors }; var execUser = ''; try { execUser = Session.getEffectiveUser().getEmail(); } catch (userErr) { execUser = '(tidak terbaca: ' + userErr.message + ')'; } var folder = null; try { folder = DriveApp.getFolderById(TARGET_FOLDER_ID); } catch (folderErr) { errors.push('Folder Drive tidak bisa diakses (' + folderErr.message + '). ' + 'Web App ini berjalan sebagai akun: "' + execUser + '". ' + 'Cek TARGET_FOLDER_ID di kode dan pastikan akun tsb punya akses ke folder tersebut.'); return { entries: entries, errors: errors }; } for (var i = 0; i < filesArray.length; i++) { var f = filesArray[i]; if (!f.base64 || !f.name) continue; try { var base64Data = f.base64.indexOf(',') > -1 ? f.base64.split(',')[1] : f.base64; var bytes = Utilities.base64Decode(base64Data); var blob = Utilities.newBlob(bytes, f.type || 'application/octet-stream', f.name); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); entries.push(f.name + ' :: ' + file.getUrl()); } catch (fileErr) { errors.push(f.name + ' (' + fileErr.message + ') [Web App berjalan sebagai: ' + execUser + ']'); } } return { entries: entries, errors: errors }; } function parseFileEntry_(line) { var parts = line.split(' :: '); return { name: parts[0] || 'file', url: parts[1] || '#' }; } // Menerima 'YYYY-MM-DD' (dari ) -> 'DD/MM/YYYY' untuk tampilan. // Kalau formatnya bukan itu (mis. sudah 'DD/MM/YYYY' atau kosong), dikembalikan apa adanya. function formatDateStr_(s) { if (!s) return ''; var parts = String(s).split('-'); if (parts.length !== 3) return s; return parts[2] + '/' + parts[1] + '/' + parts[0]; } /** * Jalankan fungsi ini SATU KALI (lewat menu Run di editor Apps Script) * setelah menempel kode versi baru ini, untuk memindahkan data lama * (format: No, Nama Depo, Uraian Kendala, Bukti Dukung, Waktu) ke * struktur kolom baru (dengan ID, Tanggal Kendala, RTL, dst). * Aman dijalankan berkali-kali — kalau sheet sudah format baru, fungsi * ini berhenti tanpa mengubah apa-apa. */ function migrateSheetToV2() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var lastRow = sheet.getLastRow(); if (lastRow === 0) { sheet.appendRow(HEADER); sheet.getRange(1, 1, 1, HEADER.length).setFontWeight('bold'); Logger.log('Sheet kosong, header baru dibuat.'); return; } var currentHeader = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (currentHeader[0] === 'ID') { Logger.log('Sheet sudah dalam format baru, migrasi dilewati.'); return; } var numDataRows = lastRow - 1; var oldData = numDataRows > 0 ? sheet.getRange(2, 1, numDataRows, 5).getValues() : []; sheet.clear(); sheet.appendRow(HEADER); sheet.getRange(1, 1, 1, HEADER.length).setFontWeight('bold'); var newRows = oldData.map(function (r) { // format lama: [No, Nama Depo, Uraian Kendala, Bukti Dukung, Waktu] return [Utilities.getUuid(), r[0], r[1], '', r[2], r[3], '', '', '', r[4]]; }); if (newRows.length > 0) { sheet.getRange(2, 1, newRows.length, HEADER.length).setValues(newRows); } Logger.log('Migrasi selesai: ' + newRows.length + ' baris dipindahkan ke format baru.'); } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
